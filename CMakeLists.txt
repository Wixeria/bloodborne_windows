# Portable runtime/CPU loader, with an optional complete Windows game renderer.
# Linux retains its existing gpu/CMakeLists.txt and build.sh workflow.
cmake_minimum_required(VERSION 3.24)
project(bbport_runtime LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" OR MSVC)
    message(FATAL_ERROR "The guest SysV ABI requires GCC or GNU-target Clang; on Windows use LLVM-MinGW x86_64.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "bbport requires a 64-bit compiler.")
endif()
if(WIN32 AND NOT MINGW)
    message(FATAL_ERROR "The Windows runtime currently requires a MinGW target with winpthreads.")
endif()

option(BB_LTO "Link-time optimization of the native runtime and renderer" OFF)
if(BB_LTO)
    enable_language(CXX)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT bb_runtime_ipo_ok OUTPUT bb_runtime_ipo_error LANGUAGES C CXX)
    if(NOT bb_runtime_ipo_ok)
        message(FATAL_ERROR "Requested LTO is unavailable: ${bb_runtime_ipo_error}")
    endif()
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
endif()
